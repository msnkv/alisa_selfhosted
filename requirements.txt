#   .venv\Scripts\python.exe -m pip install torch==2.11.0 torchaudio --index-url https://download.pytorch.org/whl/cu128
#   .venv\Scripts\python.exe -m pip install -r requirements.txt

# --- Шаг 1: мозг и текстовый режим ---
openai==3.36.1         # OpenAI-совместимый клиент, им же говорим с DeepSeek
pyyaml==6.0.3          # config.yaml
python-dotenv==1.2.4   # .env с ключом

# --- Шаг 2: интернет ---
ddgs==9.16.0           # поиск: опрашивает несколько поисковиков, без ключей
trafilatura==2.3.1     # вытаскивает читаемый текст из html, включая таблицы

# --- Шаг 4: голос ассистента ---
# torch ставится отдельной командой (см. выше). ВАЖНО: сборка под CUDA,
# и версия выбрана не произвольно.
#   - cu128, потому что Blackwell (sm_120) поддерживается начиная с
#     CUDA 12.8. На cu126 карта просто не заработает.
#   - НЕ cu130, хотя там есть версия посвежее: CUDA 13 требует драйвер
#     от 580, а на этой машине 576.88.
#   - Отсюда и 2.11.0: это самое новое, что есть в индексе cu128.
# Раньше тут стояла сборка под процессор, и это было осознанно: whisper
# работает через CTranslate2, торчу видеопамять была не нужна. Поменяли
# ради F5-TTS, которому без карты нечего делать - на процессоре у него
# минуты на фразу. Silero и whisper после замены проверены, работают.
omegaconf==2.4.0       # нужен загрузчику моделей Silero
sounddevice==0.5.6     # вывод в наушники и запись с микрофона
soundfile==0.14.0      # сохранение образцов голосов в wav
# 1.26, а не 2.x: f5-tts тянет его за собой и откатывает numpy сам.
# Насильно поднимать обратно не надо - проверено, что на 1.26 работают и
# Silero, и whisper, и wake word, и поиск.
numpy==1.26.4

# --- Шаг 5: уши ---
faster-whisper==1.2.1  # распознавание речи через CTranslate2
# ctranslate2 не ниже 4.7: в более старых int8 падал на Blackwell (sm_120)
ctranslate2==4.8.3
# cuDNN и cuBLAS пакетами, чтобы не ставить CUDA Toolkit целиком.
# Их dll лежат внутри site-packages, и Windows их сама не находит -
# каталоги регистрирует _register_cuda_dlls в assistant/stt.py
nvidia-cublas-cu12
nvidia-cudnn-cu12

# --- Шаг 6: wake word ---
# На Windows работает только через onnxruntime: tflite не поддерживается,
# а библиотека по умолчанию просит именно его - см. inference_framework
# в assistant/wake.py. Модели (~5 МБ) докачиваются при первом запуске
# в resources/models самого openwakeword, то есть внутрь .venv
openwakeword==0.6.0

# --- Этап 2: выразительный синтез (необязательно) ---
# F5-TTS - модель клонирования голоса: ей нужен образец речи и его
# расшифровка. Живее Silero, но в разы медленнее: полторы-две секунды на
# фразу против семи сотых. Поэтому держим оба движка, см. CLAUDE.md.
# Русские веса: hotstone228/F5-TTS-Russian, конфигурация F5TTS_Base,
# лицензия CC BY-NC-SA 4.0. Скачиваются сами при первом запуске.
#
# FFmpeg ставить НЕ надо, хотя документация F5 его требует: звук он
# читает ровно в одном месте, и у нас эта загрузка подменена на
# soundfile. Подробности в JOURNAL.md.
f5-tts==1.1.22
