# Что ставить на СПУТНИК — отдельную машину с микрофоном и колонкой.
#
# Мозгов тут нет и быть не должно: распознавание речи, модель и синтез
# живут на ПК, спутник только слышит кодовую фразу, пишет фразу,
# отправляет её по домашней сети и играет присланный ответ.
#
# Поэтому здесь НЕТ: openai, faster-whisper, ctranslate2, torch,
# openwakeword, ddgs, trafilatura. Проверено запуском с запретом на эти
# пакеты — спутник собирается и работает без них.
#
# Установка на спутнике:
#   python -m venv .venv
#   .venv/bin/python -m pip install -r requirements-client.txt
#   скопировать туда же папку assistant/ и config.yaml
#   в config.yaml поставить server.url с адресом ПК и mic.vad: energy
#   запуск:  .venv/bin/python -m assistant.cli --client
#
# Модель кодовой фразы (сорок четыре мегабайта) скачается сама при
# первом запуске.

numpy==1.26.4          # звук и кадры
sounddevice==0.5.6     # микрофон и динамик
pyyaml==6.0.3          # config.yaml
python-dotenv==1.2.4   # .env, хотя ключ спутнику и не нужен
vosk==0.3.45           # кодовая фраза по-русски

# Необязательно: torch добавляет silero-vad, он точнее определяет конец
# фразы, чем определение по громкости. Полтора гигабайта ради одной
# маленькой модели — на спутнике обычно не стоит того, поэтому по
# умолчанию mic.vad: energy. Если машина не слабая и хочется точнее:
#   pip install torch --index-url https://download.pytorch.org/whl/cpu
